# SIBO devices booted with an external SSD pack attached.
# Run via `scripts/test-boot.sh --ssd`. Format:
#   id | rom-filename | boot-seconds | min-variance | extra-args | device-id
#
# `id` is the test name (used for golden image filenames in tests/golden/);
# `device-id` is the actual factory id passed to --device (so we can run
# the same physical device with and without an SSD attached).
#
# Two fixtures are used:
#   - blank-128k-ram.ssd: zero-filled 128 KiB attached as a RAM pack
#     (info byte 0x03). EPOC16 detects it and reports the drive
#     "Unformatted" until it is formatted on the device (System screen,
#     Disk -> Format disk). The row verifies the SIBO slot-scan path
#     doesn't disturb boot.
#   - fefs-128k.ssd: a FEFS24 Flash pack with files, generated by
#     frontend/src/lib/fefs.ts and attached as a Type 1 Flash pack (info
#     byte 0x23) — the configuration EPOC16 mounts as a browsable
#     read/write drive. Regenerate with:
#       node --experimental-strip-types --input-type=module -e "
#         import { createFlashPack, addFileToPack } from './frontend/src/lib/fefs.ts';
#         import { writeFileSync } from 'node:fs';
#         let p = createFlashPack(0x20000, 'FIXTURE');
#         p = addFileToPack(p, 'HELLO.TXT', new TextEncoder().encode('Hello from tests/fixtures/fefs-128k.ssd\n'));
#         p = addFileToPack(p, 'REPORT.WRD', new Uint8Array(1000).fill(0x42), 'WRD');
#         writeFileSync('tests/fixtures/fefs-128k.ssd', p);"
